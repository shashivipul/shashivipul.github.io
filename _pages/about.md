Namaste! I am **Vipul**, an AI researcher working on trustworthiness and scalability challenges in **Graph Machine Learning**, with a particular focus on fairness and robustness. I am currently a fifth-year PhD candidate in the Department of Electrical Engineering at the [Indian Institute of Technology Delhi](https://home.iitd.ac.in). My work spans graph deep learning, graph coarsening, medical image analytics, financial time-series, and knowledge distillation. Recently, I've been exploring the trustworthiness of LLMs and graph-language models. My research is advised by [Prof. Sandeep Kumar](https://sites.google.com/view/sandeepkr/home) and [Prof. Jayadeva](https://web.iitd.ac.in/~jayadeva/) at the [Machine Intelligence Signals and Networks (MISN) Lab](https://misn.iitd.ac.in). I am incredibly grateful and excited to mention that I have received the prestigious [Prime Minister’s Research Fellowship (PMRF)](https://www.pmrf.in) as part of my PhD journey.

I have joined [Fujitsu Research](https://global.fujitsu/en-apac) as an intern in the AI lab, where I will be exploring the area of multi-agent systems, knowledge graphs, and dynamic graphs.

I was a Visiting Junior Fellow in the lab of [Prof. Tsuyoshi Murata](https://www.net.c.titech.ac.jp/murata.html) at the [Institute of Science Tokyo](https://www.isct.ac.jp/en) from April to August 2026. Before starting my PhD, I completed my **M.Tech.** in the Department of Electrical Engineering at the [Indian Institute of Technology Patna](https://www.iitp.ac.in), where I worked on biomedical signal and image processing under the supervision of [Prof. Maheshkumar H. Kolekar](https://www.iitp.ac.in/~mahesh/). I earned my **B.Tech.** in Electronics and Communication Engineering from the [Dr. A.P.J. Abdul Kalam Technical University (AKTU)](https://aktu.ac.in).

(function () {
  var aff = document.querySelector('.affiliations');
  if (!aff) return;
  var root = aff.parentNode;
  var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* Mark pieces of the page so the styles can target them */
  root.classList.add('home');
  var social = root.querySelector('.fa-linkedin');
  social = social && social.closest('div');
  if (social) social.classList.add('home-social');
  aff.querySelectorAll('.affil').forEach(function (a) {
    if (/present/i.test(a.querySelector('.affil__years').textContent)) a.classList.add('is-current');
  });
  var updates = Array.prototype.find.call(root.querySelectorAll('h2'), function (h) { return /updates/i.test(h.textContent); });
  var updList = updates && updates.nextElementSibling;
  if (updList && updList.tagName === 'UL' && updList.firstElementChild) {
    var badge = document.createElement('span');
    badge.className = 'home-new';
    badge.textContent = 'NEW';
    updList.firstElementChild.appendChild(badge);
  }

  if (reduce || !('IntersectionObserver' in window)) return;
  root.classList.add('home--animate');

  /* Stagger groups of siblings */
  function stagger(list, step, base) {
    Array.prototype.forEach.call(list, function (el, i) {
      el.classList.add('reveal');
      el.dataset.delay = (base || 0) + i * step + 'ms';
    });
  }
  stagger(Array.prototype.filter.call(root.children, function (el) {
    return el.tagName === 'P' && !el.closest('.affiliations');
  }), 120);
  stagger(aff.querySelectorAll('.affil'), 110, 250);
  if (social) stagger(social.querySelectorAll('a'), 80, 150);
  if (updates) stagger([updates], 0);
  if (updList) stagger(updList.children, 60);

  /* Play the reveal every time an element enters the screen, and reset it
     when it leaves, so it animates again on scroll up or down */
  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (e) {
      var el = e.target;
      clearTimeout(el._revealTimer);
      if (e.isIntersecting) {
        el.style.transitionDelay = el.dataset.delay || '0ms';
        el.classList.add('is-visible');
        /* drop the delay afterwards so hover effects respond instantly */
        el._revealTimer = setTimeout(function () { el.style.transitionDelay = ''; }, 1600);
      } else {
        el.style.transitionDelay = '0ms';
        el.classList.remove('is-visible');
      }
    });
  }, { threshold: 0.1, rootMargin: '0px 0px -30px 0px' });
  root.querySelectorAll('.reveal').forEach(function (el) { io.observe(el); });

  /* Highlighter sweep on bold keywords, replayed whenever the intro comes back into view */
  var hlTimer;
  var hl = new IntersectionObserver(function (entries) {
    clearTimeout(hlTimer);
    if (entries[entries.length - 1].isIntersecting) {
      hlTimer = setTimeout(function () { root.classList.add('home--highlight'); }, 500);
    } else {
      root.classList.remove('home--highlight');
    }
  }, { threshold: 0.2 });
  var firstP = root.querySelector(':scope > p');
  if (firstP) hl.observe(firstP);
})();
</script>
